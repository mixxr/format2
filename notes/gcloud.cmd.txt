# to create a docker image
gcloud beta run jobs deploy webclaw-job --source=. --execute-now --region=europe-west1 \
--add-volume name=gcs-1,bucket=rws-data,type=cloud-storage \
--add-volume-mount volume=gcs-1,mount-path=/data \
--env-vars-file=./gcloud.dev.env 

# Create a scheduler job that triggers the Cloud Run job daily at 17 UTC+2
gcloud scheduler jobs create http daily-webclaw-schedule \
  --location=europe-west1 \
  --schedule="0 17 * * MON-FRI" \
  --uri="https://europe-west1-run.googleapis.com/apis/run.googleapis.com/v1/namespaces/invcerts/jobs/webclaw-job:run" \
  --http-method=POST \
  --oauth-service-account-email=600125851897-compute@developer.gserviceaccount.com \
  --oauth-token-scope="https://www.googleapis.com/auth/cloud-platform" \
  --time-zone="Europe/Rome" \
  --attempt-deadline="3m" \
  --description="Triggers daily webclaw Cloud Run job"

# read config scheduler
gcloud scheduler jobs describe daily-webclaw-schedule \
  --location=europe-west1
